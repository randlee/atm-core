API per the beads-cache README: `Cache(repo, scopes=[...])`, `get(id)`, and `write(id, op, args, precondition, idempotency_key, actor)` with ops close and update.

## Delivery of bd_cache_rs

atm-bd-types 0.2.0 does **not** depend on bd-cache-rs in Cargo or `Requires-Dist` (PyPI rejects URL dependencies). `CacheStore` imports `bd_cache_rs` at call time (`py.import`); a failure is `CACHE_UNAVAILABLE`, naming the wheel. The bo-10 hooks declare both packages in their PEP 723 header, pinning bd-cache-rs with `[tool.uv.sources]`, one URL per platform marker. Platforms without a wheel (Intel macOS, Linux arm64, Windows) get `CACHE_UNAVAILABLE`.

```python
# atm-bd-types 0.2.0 Python surface: exactly the names the three lite hooks use.
import enum
import typing

class Status(enum.Enum):
    OPEN = ...
    IN_PROGRESS = ...
    BLOCKED = ...
    DEFERRED = ...
    CLOSED = ...

class AssignTransition(enum.Enum):
    ASSIGN_SAME = ...
    ASSIGN = ...
    ASSIGN_CLOSED = ...
    ASSIGN_NOT_READY = ...
    @staticmethod
    def select(status: Status, assignee: str | None, wanted: str) -> AssignTransition: ...

class StartTransition(enum.Enum):
    START = ...
    START_AGAIN = ...
    START_CLOSED = ...
    START_NOT_READY = ...
    @staticmethod
    def select(status: Status) -> StartTransition: ...

class CloseTransition(enum.Enum):
    GIVE_BACK = ...
    WORK_CLOSE = ...
    QA_PASS = ...
    QA_FINDINGS = ...
    @staticmethod
    def select(req: HookRequest, bead: Bead) -> CloseTransition | Refusal:
        r"""outcome x stage (x verdict kind for QA); REQUEST_INVALID / QA_VERDICT when they do not decode."""

class Template(enum.Enum):
    REFUSE = ...
    DEV_CLOSE = ...
    QA_PASS = ...
    QA_FINDINGS = ...

class ErrorId(enum.Enum):
    REQUEST_INVALID = ...
    PHASE_UNREADABLE = ...
    PROFILE_MISMATCH = ...
    CACHE_UNAVAILABLE = ...
    BEAD_NOT_FOUND = ...
    BEAD_NOT_READY = ...
    BEAD_CLOSED = ...
    BEAD_RACED = ...
    ASSIGN_NOT_LEAD = ...
    TASK_NOT_ASSIGNEE = ...
    CLOSE_REASON_EMPTY = ...
    CLOSE_REPORT_NOT_INLINE = ...
    STACK_NOT_RECORDED = ...
    QA_VERDICT = ...
    QA_NOT_ON_STACK = ...
    QA_NOTHING_TO_REVIEW = ...
    FINDINGS_NOT_POURED = ...

@typing.final
class Refusal:
    def __new__(cls, id: ErrorId, **context: str | list[str]) -> Refusal: ...
    @property
    def id(self) -> ErrorId: ...

@typing.final
class Layer:
    @property
    def branch(self) -> str: ...
    @property
    def head(self) -> str: ...

@typing.final
class Finding:
    @property
    def id(self) -> str: ...
    @property
    def blocking(self) -> bool: ...

@typing.final
class QaPass:
    @property
    def sha(self) -> str: ...

@typing.final
class QaFail:
    @property
    def sha(self) -> str: ...
    @property
    def findings(self) -> list[Finding]: ...

@typing.final
class Bead:
    @property
    def id(self) -> str: ...
    @property
    def status(self) -> Status: ...
    @property
    def assignee(self) -> str | None: ...
    @property
    def open_blockers(self) -> list[str]: ...
    @property
    def branch(self) -> str: ...
    @property
    def stack_bead(self) -> str: ...
    @property
    def sprint(self) -> str: ...
    @property
    def phase(self) -> str: ...
    def blocks(self, bead_id: str) -> bool: ...
    def layer_for(self, branch: str) -> Layer | None:
        r"""On a stack bead: its ledger layer for `branch`."""

@typing.final
class PhaseFile:
    @property
    def lead(self) -> str: ...
    def template(self, which: Template) -> str: ...

@typing.final
class BeadPatch:
    r"""One guarded write: applied only if status and assignee still equal the snapshot's, else BEAD_RACED."""
    @staticmethod
    def assign(snapshot: Bead, agent: str) -> BeadPatch: ...
    @staticmethod
    def start(snapshot: Bead) -> BeadPatch: ...
    @staticmethod
    def reopen(snapshot: Bead) -> BeadPatch: ...
    @staticmethod
    def close(snapshot: Bead, evidence: Layer | QaPass | QaFail) -> BeadPatch: ...

@typing.final
class HookRequest:
    @staticmethod
    def parse(stdin: str) -> HookRequest | Refusal: ...
    def is_phase_task(self) -> bool: ...
    @property
    def phase(self) -> str: ...
    @property
    def phase_config_path(self) -> str: ...
    @property
    def caller(self) -> str: ...
    @property
    def task_id(self) -> str: ...
    @property
    def assignee(self) -> str | None: ...
    @property
    def close_report(self) -> str | None: ...
    @property
    def close_reason(self) -> str | None: ...
    def qa_pass(self) -> QaPass | Refusal: ...
    def qa_findings(self) -> QaFail | Refusal: ...

@typing.final
class HookAction:
    @staticmethod
    def close(template: str, vars: Layer | QaPass | QaFail | str) -> HookAction:
        r"""atm task close --template <template> --vars <vars>; str is the give-back reason."""

@typing.final
class HookReply:
    @staticmethod
    def not_applicable() -> HookReply: ...
    @staticmethod
    def pass_through() -> HookReply: ...
    @staticmethod
    def refused(refusal: Refusal) -> HookReply: ...
    @staticmethod
    def handled(actions: list[HookAction]) -> HookReply: ...
    def to_json(self) -> str: ...

class BeadStore(typing.Protocol):
    def show(self, bead_id: str) -> Bead | Refusal: ...
    def target_layer(self, qa_bead: Bead) -> Layer | Refusal:
        r"""The ledger layer of the QA bead's closed dev/fix blocker, or of its checked_bead."""
    def update(self, patch: BeadPatch, actor: str) -> Refusal | None: ...

class PhaseFiles(typing.Protocol):
    def load(self, path: str, profile: str) -> PhaseFile | Refusal: ...

@typing.final
class CacheStore(BeadStore):
    @staticmethod
    def for_phase(phase: str) -> CacheStore | Refusal: ...
    def show(self, bead_id: str) -> Bead | Refusal: ...
    def target_layer(self, qa_bead: Bead) -> Layer | Refusal: ...
    def update(self, patch: BeadPatch, actor: str) -> Refusal | None: ...

@typing.final
class PhaseFilesCli(PhaseFiles):
    def __new__(cls) -> PhaseFilesCli: ...
    def load(self, path: str, profile: str) -> PhaseFile | Refusal: ...
```
